cmake_minimum_required(VERSION 3.22)
project(MuckeBabyLinux LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(VLC REQUIRED IMPORTED_TARGET libvlc>=3.0)
if(VLC_VERSION VERSION_GREATER_EQUAL 4.0)
    message(FATAL_ERROR "Dieser Port benötigt libVLC 3.x")
endif()
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
find_package(nlohmann_json 3.10 REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../Sources/Models.swift")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Sources/Models.swift" MODEL_SOURCE)
string(REGEX MATCH "static let version = \"([0-9]+\\.[0-9]+\\.[0-9]+)\"" VERSION_MATCH "${MODEL_SOURCE}")
if(NOT VERSION_MATCH)
    message(FATAL_ERROR "AppInfo.version fehlt")
endif()
set(MUCKE_VERSION "${CMAKE_MATCH_1}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../Resources/seed-stations.example.json")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../Resources/seed-stations.example.json" MUCKE_PUBLIC_SEED_HASH)
configure_file(src/version.h.in generated/version.h @ONLY)
add_compile_options(-Wall -Wextra -Wpedantic "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=." "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build")
file(GLOB DATA_SOURCES CONFIGURE_DEPENDS src/data/*.cpp src/data/*.h)
file(GLOB EXPORT_SOURCES CONFIGURE_DEPENDS src/export/*.cpp src/export/*.h)
file(GLOB MEDIA_SOURCES CONFIGURE_DEPENDS src/media/*.cpp src/media/*.h)
file(GLOB UI_SOURCES CONFIGURE_DEPENDS src/ui/*.cpp src/ui/*.h)
file(GLOB INTEGRATION_SOURCES CONFIGURE_DEPENDS src/integration/*.cpp src/integration/*.h)
add_library(mucke-data STATIC src/types.h src/types.cpp ${DATA_SOURCES})
target_include_directories(mucke-data PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(mucke-data PUBLIC Qt6::Core Qt6::Network nlohmann_json::nlohmann_json)
add_library(mucke-export STATIC ${EXPORT_SOURCES})
target_link_libraries(mucke-export PUBLIC mucke-data)
add_library(mucke-media STATIC ${MEDIA_SOURCES})
target_link_libraries(mucke-media PUBLIC mucke-data mucke-export PkgConfig::VLC PkgConfig::PULSE)
add_library(mucke-ui STATIC ${UI_SOURCES})
target_link_libraries(mucke-ui PUBLIC mucke-data mucke-media Qt6::Widgets)
add_library(mucke-resources OBJECT resources.qrc)
target_link_libraries(mucke-resources PRIVATE Qt6::Core)
add_executable(mucke-baby src/main.cpp $<TARGET_OBJECTS:mucke-resources> ${INTEGRATION_SOURCES})
target_link_libraries(mucke-baby PRIVATE mucke-ui Qt6::DBus)
install(TARGETS mucke-baby RUNTIME DESTINATION bin)
install(FILES packaging/de.danielmuller.muckebaby.desktop DESTINATION share/applications)
install(FILES ../icons/app-icon-s123.png DESTINATION share/icons/hicolor/1024x1024/apps RENAME mucke-baby.png)
install(FILES ../LICENSE ../THIRD-PARTY.md ../docs/linux.md packaging/nlohmann-json-LICENSE.MIT DESTINATION share/doc/mucke-baby)
option(MUCKE_BUILD_TESTS "Linux-Regressionstests bauen" ON)
if(MUCKE_BUILD_TESTS)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    enable_testing()
    add_executable(recording_codec_probe tests/recording_codec_probe.cpp)
    target_link_libraries(recording_codec_probe PRIVATE mucke-media)
    add_executable(library_interop tests/library_interop.cpp $<TARGET_OBJECTS:mucke-resources>)
    target_link_libraries(library_interop PRIVATE mucke-data)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME installer_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/installer_test.py)
    file(GLOB TEST_SOURCES CONFIGURE_DEPENDS tests/*_test.cpp tests/*_tests.cpp tests/*_smoke.cpp tests/*_measurement.cpp)
    foreach(TEST_SOURCE IN LISTS TEST_SOURCES)
        get_filename_component(TEST_NAME "${TEST_SOURCE}" NAME_WE)
        if(TEST_NAME STREQUAL "ui_song_export_test")
            add_executable(${TEST_NAME} "${TEST_SOURCE}" src/ui/songexportdialog.cpp src/media/mediaengine.h)
            target_include_directories(${TEST_NAME} PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
            target_link_libraries(${TEST_NAME} PRIVATE Qt6::Widgets Qt6::Test nlohmann_json::nlohmann_json)
            add_test(NAME ${TEST_NAME} COMMAND ${TEST_NAME})
            set_tests_properties(${TEST_NAME} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
            continue()
        endif()
        add_executable(${TEST_NAME} "${TEST_SOURCE}" $<TARGET_OBJECTS:mucke-resources>)
        target_link_libraries(${TEST_NAME} PRIVATE mucke-ui Qt6::Test Qt6::DBus)
        if(TEST_NAME STREQUAL "mpris_test")
            find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
            target_sources(${TEST_NAME} PRIVATE src/integration/mpris.cpp src/integration/mpris.h)
            add_test(NAME ${TEST_NAME} COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:${TEST_NAME}>)
            set_tests_properties(${TEST_NAME} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
            continue()
        endif()
        if(TEST_NAME STREQUAL "resource_test")
            target_compile_definitions(${TEST_NAME} PRIVATE MUCKE_PUBLIC_SEED_HASH="${MUCKE_PUBLIC_SEED_HASH}")
        endif()
        if(NOT TEST_NAME MATCHES "_smoke$|media_pulse_tests|_measurement$|_pcm$|^ui_song_backend_test$")
            add_test(NAME ${TEST_NAME} COMMAND ${TEST_NAME})
            set_tests_properties(${TEST_NAME} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
        endif()
    endforeach()
endif()
set(CPACK_PACKAGE_NAME mucke-baby)
set(CPACK_PACKAGE_VERSION "${MUCKE_VERSION}")
set(CPACK_PACKAGE_VENDOR "Mucke, Baby!")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native internet radio with stream recording and seven themes")
set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGE_FILE_NAME "mucke-baby-${MUCKE_VERSION}-linux-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
include(CPack)
